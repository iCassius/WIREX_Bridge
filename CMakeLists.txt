cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(wirex_dmrd LANGUAGES CXX)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(wirex_dmrd src/dmrd.cpp)
target_include_directories(wirex_dmrd PUBLIC include)
target_compile_features(wirex_dmrd PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(wirex_dmrd PRIVATE /W4 /permissive-)
else()
    target_compile_options(wirex_dmrd PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(dmrd_test tests/dmrd_test.cpp)
target_link_libraries(dmrd_test PRIVATE wirex_dmrd)

add_library(wirex_direct_frame src/direct_frame.cpp)
target_include_directories(wirex_direct_frame PUBLIC include)
target_compile_features(wirex_direct_frame PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(wirex_direct_frame PRIVATE /W4 /permissive-)
else()
    target_compile_options(wirex_direct_frame PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(direct_frame_test tests/direct_frame_test.cpp)
target_link_libraries(direct_frame_test PRIVATE wirex_direct_frame)

add_library(wirex_direct_stream src/direct_stream.cpp)
target_include_directories(wirex_direct_stream PUBLIC include)
target_compile_features(wirex_direct_stream PUBLIC cxx_std_17)
target_link_libraries(wirex_direct_stream PUBLIC wirex_direct_frame)

if(MSVC)
    target_compile_options(wirex_direct_stream PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(wirex_direct_frame PRIVATE /utf-8)
else()
    target_compile_options(wirex_direct_stream PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(direct_stream_test tests/direct_stream_test.cpp)
target_link_libraries(direct_stream_test PRIVATE wirex_direct_stream)
if(MSVC)
    target_compile_options(direct_stream_test PRIVATE /W4 /permissive- /utf-8)
endif()

add_library(wirex_wave_file src/wave_file.cpp)
target_include_directories(wirex_wave_file PUBLIC include)
target_compile_features(wirex_wave_file PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(wirex_wave_file PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(wirex_wave_file PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(wave_file_test tests/wave_file_test.cpp)
target_link_libraries(wave_file_test PRIVATE wirex_wave_file)
if(MSVC)
    target_compile_options(wave_file_test PRIVATE /W4 /permissive- /utf-8)
endif()

add_library(wirex_pcm_gain src/pcm_gain.cpp)
target_include_directories(wirex_pcm_gain PUBLIC include)
target_compile_features(wirex_pcm_gain PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(wirex_pcm_gain PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(wirex_pcm_gain PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(pcm_gain_test tests/pcm_gain_test.cpp)
target_link_libraries(pcm_gain_test PRIVATE wirex_pcm_gain)
if(MSVC)
    target_compile_options(pcm_gain_test PRIVATE /W4 /permissive- /utf-8)
endif()

if(WIN32)
    add_executable(wirex_probe src/probe_main.cpp)
    target_compile_features(wirex_probe PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(wirex_probe PRIVATE /W4 /permissive- /utf-8)
    endif()
    target_link_libraries(wirex_probe PRIVATE wirex_direct_stream wirex_direct_frame wirex_wave_file wirex_pcm_gain winmm)
endif()

enable_testing()
add_test(NAME dmrd_test COMMAND dmrd_test)
add_test(NAME direct_frame_test COMMAND direct_frame_test)
add_test(NAME direct_stream_test COMMAND direct_stream_test)
add_test(NAME wave_file_test COMMAND wave_file_test)
add_test(NAME pcm_gain_test COMMAND pcm_gain_test)
